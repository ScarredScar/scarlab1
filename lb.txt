section .data
    ; Объявляем строку в памяти. 10 — это код символа новой строки (\n)
    message db "System programming test message", 10
    ; Автоматически вычисляем длину строки в байтах
    ; (текущий адрес минус адрес начала строки)
    message_len equ $ - message

section .text
    global _start       ; Точка входа в программу для компоновщика

_start:
    ; --- ПОДГОТОВКА СИСТЕМНОГО ВЫЗОВА SYS_WRITE ---
    mov rax, 1           ; Код 1 — операция sys_write (запись в поток)
    mov rdi, 1           ; Код 1 — вывод в консоль (stdout)
    mov rsi, message     ; Адрес строки в регистр RSI
    mov rdx, message_len ; Длина строки в регистр RDX
    syscall              ; Переключение в Ring 0, передача управления ОС

    ; --- ПОДГОТОВКА СИСТЕМНОГО ВЫЗОВА SYS_EXIT ---
    ; Если программу не завершить корректно,
    ; процессор пойдёт выполнять случайный мусор в памяти
    mov rax, 60          ; Код 60 — операция sys_exit (завершение программы)
    mov rdi, 0           ; Код возврата 0 (ошибок нет)
    syscall              ; Передача управления ОС для уничтожения процесса
